```go
package ssh

import (
	"encoding/json"
	"fmt"
	"net"
	"os"
	"path/filepath"
	"sync"
	"time"

	gossh "golang.org/x/crypto/ssh"
)

// HostEntry 是 known_hosts 里的一台 SSH 主机(链上任一节点:跳板或终点)。
type HostEntry struct {
	Host        string    `json:"host"`         // "host:port"
	Fingerprint string    `json:"fingerprint"`  // SHA256:base64
	Algorithm   string    `json:"algorithm"`    // 如 ssh-ed25519
	AddedAt     time.Time `json:"added_at"`
}

// KnownHosts 是 TOFU 指纹库:首连写入,复连校验,不匹配拒连。
type KnownHosts struct {
	mu   sync.Mutex
	path string
}

func NewKnownHosts(path string) *KnownHosts {
	return &KnownHosts{path: path}
}

// CallbackFor 返回指定 host:port 的 TOFU 校验回调。
// x/crypto 会把 hostname 的端口剥掉再传进来,所以 key 由调用方(链构建器,
// 知道每个节点的准确 host:port)闭包注入,不从参数拼。
func (k *KnownHosts) CallbackFor(hostport string) gossh.HostKeyCallback {
	return func(_ string, _ net.Addr, key gossh.PublicKey) error {
		fp := gossh.FingerprintSHA256(key)
		k.mu.Lock()
		defer k.mu.Unlock()

		entries, err := k.loadLocked()
		if err != nil {
			return fmt.Errorf("load known_hosts: %w", err)
		}
		for _, e := range entries {
			if e.Host != hostport {
				continue
			}
			if e.Fingerprint == fp {
				return nil
			}
			return fmt.Errorf("ssh host key mismatch for %s: 已知 %s,实际 %s(可能是服务器重装或中间人;可在仪表盘「服务器指纹」删除该条目后重连)", hostport, e.Fingerprint, fp)
		}
		entries = append(entries, HostEntry{
			Host:        hostport,
			Fingerprint: fp,
			Algorithm:   key.Type(),
			AddedAt:     time.Now(),
		})
		return k.saveLocked(entries)
	}
}

func (k *KnownHosts) List() []HostEntry {
	k.mu.Lock()
	defer k.mu.Unlock()
	entries, err := k.loadLocked()
	if err != nil {
		return nil
	}
	return entries
}

func (k *KnownHosts) Remove(hostport string) error {
	k.mu.Lock()
	defer k.mu.Unlock()
	entries, err := k.loadLocked()
	if err != nil {
		return err
	}
	kept := entries[:0]
	for _, e := range entries {
		if e.Host != hostport {
			kept = append(kept, e)
		}
	}
	return k.saveLocked(kept)
}

func (k *KnownHosts) Clear() error {
	k.mu.Lock()
	defer k.mu.Unlock()
	return k.saveLocked(nil)
}

func (k *KnownHosts) loadLocked() ([]HostEntry, error) {
	data, err := os.ReadFile(k.path)
	if os.IsNotExist(err) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	var entries []HostEntry
	if err := json.Unmarshal(data, &entries); err != nil {
		return nil, err
	}
	return entries, nil
}

// saveLocked 原子写:临时文件+rename,避免半行写入导致指纹库损坏。
func (k *KnownHosts) saveLocked(entries []HostEntry) error {
	if err := os.MkdirAll(filepath.Dir(k.path), 0o755); err != nil {
		return err
	}
	data, err := json.MarshalIndent(entries, "", "  ")
	if err != nil {
		return err
	}
	tmp := k.path + ".tmp"
	if err := os.WriteFile(tmp, data, 0o600); err != nil {
		return err
	}
	return os.Rename(tmp, k.path)
}
```

```go
// Package sshtest 提供同进程 loopback SSH 服务器,供 ssh outbound 全链路测试。
// 仅测试 import,不进生产二进制。
package sshtest

import (
	"crypto/ed25519"
	"crypto/rand"
	"encoding/binary"
	"fmt"
	"net"
	"strconv"
	"sync"
	"testing"

	gossh "golang.org/x/crypto/ssh"
)

// Server 是一台内存 sshd。Password/AuthorizedKey 二选一或并存;
// NextDial 决定 direct-tcpip 的落点(默认真连目标)。
type Server struct {
	t            *testing.T
	Listener     net.Listener
	HostKey      gossh.Signer
	Password     string
	Authorized   gossh.PublicKey
	Username     string
	// NextDial 非空时,通道目标经它转发(多跳级联用:下一跳 sshd)。
	NextDial     func(addr string) (net.Conn, error)

	mu       sync.Mutex
	chClosed []func()
}

// Start 在 127.0.0.1 随机端口起一台 sshd,返回实例与 host:port。
func Start(t *testing.T, opts ...Option) (*Server, string) {
	t.Helper()
	s := &Server{t: t, Username: "tester"}
	for _, o := range opts {
		o(s)
	}
	_, priv, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		t.Fatalf("host key gen: %v", err)
	}
	s.HostKey, err = gossh.NewSignerFromKey(priv)
	if err != nil {
		t.Fatalf("host key signer: %v", err)
	}
	l, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatalf("listen: %v", err)
	}
	s.Listener = l
	go s.serve()
	t.Cleanup(s.Close)
	return s, l.Addr().String()
}

type Option func(*Server)

func WithPassword(p string) Option    { return func(s *Server) { s.Password = p } }
func WithUsername(u string) Option    { return func(s *Server) { s.Username = u } }
func WithAuthorized(k gossh.PublicKey) Option {
	return func(s *Server) { s.Authorized = k }
}
func WithNextDial(f func(addr string) (net.Conn, error)) Option {
	return func(s *Server) { s.NextDial = f }
}

func (s *Server) Port() int {
	return s.Listener.Addr().(*net.TCPAddr).Port
}

func (s *Server) Close() {
	_ = s.Listener.Close()
	s.mu.Lock()
	for _, f := range s.chClosed {
		f()
	}
	s.chClosed = nil
	s.mu.Unlock()
}

func (s *Server) serve() {
	for {
		c, err := s.Listener.Accept()
		if err != nil {
			return
		}
		go s.handle(c)
	}
}

func (s *Server) handle(c net.Conn) {
	cfg := &gossh.ServerConfig{}
	if s.Password != "" {
		cfg.PasswordCallback = func(conn gossh.ConnMetadata, password []byte) (*gossh.Permissions, error) {
			if string(password) != s.Password || conn.User() != s.Username {
				return nil, fmt.Errorf("auth rejected for %s", conn.User())
			}
			return &gossh.Permissions{}, nil
		}
	}
	if s.Authorized != nil {
		cfg.PublicKeyCallback = func(conn gossh.ConnMetadata, key gossh.PublicKey) (*gossh.Permissions, error) {
			if key.Type() != s.Authorized.Type() {
				return nil, fmt.Errorf("key type mismatch")
			}
			return &gossh.Permissions{}, nil
		}
	}
	sconn, chans, reqs, err := gossh.NewServerConn(c, cfg)
	if err != nil {
		return
	}
	_ = sconn
	go s.handleReqs(reqs)
	for newCh := range chans {
		if newCh.ChannelType() != "direct-tcpip" {
			_ = newCh.Reject(gossh.UnknownChannelType, "unsupported")
			continue
		}
		s.handleDirectTCPIP(newCh)
	}
}

type directTCPIPMsg struct {
	Addr       string
	Port       uint32
	OriginAddr string
	OriginPort uint32
}

func (s *Server) handleDirectTCPIP(newCh gossh.NewChannel) {
	payload := newCh.ExtraData()
	var m directTCPIPMsg
	if err := gossh.Unmarshal(payload, &m); err != nil {
		_ = newCh.Reject(gossh.Prohibited, "bad payload") // v0.48 常量名是 Prohibited
		return
	}
	target := net.JoinHostPort(m.Addr, strconv.Itoa(int(m.Port)))
	var upstream net.Conn
	var err error
	if s.NextDial != nil {
		upstream, err = s.NextDial(target)
	} else {
		upstream, err = net.Dial("tcp", target)
	}
	if err != nil {
		_ = newCh.Reject(gossh.ConnectionFailed, err.Error())
		return
	}
	ch, reqs, err := newCh.Accept()
	if err != nil {
		_ = upstream.Close()
		return
	}
	go gossh.DiscardRequests(reqs)
	go func() {
		defer upstream.Close()
		defer ch.Close()
		pipe(ch, upstream)
	}()
}

// writeCloseSink 是 ssh.Channel 与 net.Conn 共有的半关能力(io.WriteCloser
// 本身没有 CloseWrite,必须用这个窄接口)。
type writeCloseSink interface {
	io.Writer
	CloseWrite() error
}

func pipe(a io.ReadWriter, b io.ReadWriter) {
	done := make(chan struct{}, 2)
	copySide := func(dst writeCloseSink, src io.Reader) {
		_, _ = io.Copy(struct{ io.Writer }{dst}, src)
		_ = dst.CloseWrite()
		done <- struct{}{}
	}
	go copySide(b.(writeCloseSink), a)
	go copySide(a.(writeCloseSink), b)
	<-done
	<-done
}

func (s *Server) handleReqs(reqs <-chan *gossh.Request) {
	for req := range reqs {
		if req.WantReply {
			_ = req.Reply(false, nil)
		}
	}
}

// DialProxy 是 sshtest 直连目标的 stub:测试 direct-tcpip 时用它当目标
// (回显服务器)。
func EchoListener(t *testing.T) (addr string, close func()) {
	l, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatalf("echo listen: %v", err)
	}
	go func() {
		for {
			c, err := l.Accept()
			if err != nil {
				return
			}
			go func(c net.Conn) {
				defer c.Close()
				_, _ = io.Copy(c, c)
			}(c)
		}
	}()
	return l.Addr().String(), l.Close
}

var _ = binary.LittleEndian // 保留占位,避免未用 import 报错(实现时按需删)
```

```go
package ssh

import (
	"context"
	"fmt"
	"net"
	"strconv"
	"time"

	"aliang.one/nursorgate/inbound/tun/dialer"
	gossh "golang.org/x/crypto/ssh"
)

// nodeConfig 是链上一个 SSH 节点(跳板或终点)的连接参数。
type nodeConfig struct {
	hostport      string
	user          string
	password      string
	privateKey    string
	keyPassphrase string
}

func nodeFromAddr(addr, user, pass, key, keyPass string) nodeConfig {
	return nodeConfig{hostport: addr, user: user, password: pass, privateKey: key, keyPassphrase: keyPass}
}

// Config 是 outbound 的原始配置(Sec 改名自 spec §3 字段)。
type Config struct {
	User          string
	Password      string
	PrivateKey    string
	KeyPassphrase string
	PoolSize      int
	Jumps         []nodeConfig
}

// authMethods 返回节点可用认证方式;私钥解析失败返回错误(New 时即校验,
// fail-fast——运行期不再有解析路径)。
func authMethods(n nodeConfig) ([]gossh.AuthMethod, error) {
	var methods []gossh.AuthMethod
	if n.password != "" {
		methods = append(methods, gossh.Password(n.password))
		methods = append(methods, gossh.KeyboardInteractive(
			func(_ string, _ string, questions []string, _ []bool) ([]string, error) {
				answers := make([]string, len(questions))
				for i := range answers {
					answers[i] = n.password
				}
				return answers, nil
			}))
	}
	if n.privateKey != "" {
		var signer gossh.Signer
		var err error
		if n.keyPassphrase != "" {
			signer, err = gossh.ParsePrivateKeyWithPassphrase([]byte(n.privateKey), []byte(n.keyPassphrase))
		} else {
			signer, err = gossh.ParsePrivateKey([]byte(n.privateKey))
		}
		if err != nil {
			return nil, fmt.Errorf("parse private key: %w", err)
		}
		methods = append(methods, gossh.PublicKeys(signer))
	}
	return methods, nil

func newChainDialer(cfg Config, hosts *KnownHosts) *chainDialer {

	d.nodes = append(d.nodes, cfg.Jumps...)
	d.nodes = append(d.nodes, nodeConfig{

const handshakeTimeout = 15 * time.Second

// dialChain 建链:第 0 跳 TCP 直连;后续跳经前一跳的 direct-tcpip 到下一跳
// 的 sshd,再在其上跑 NewClientConn——即 OpenSSH ProxyJump 语义。
func (d *chainDialer) dialChain(ctx context.Context) (*gossh.Client, error) {
	var client *gossh.Client
	for i, n := range d.nodes {
		if err := ctx.Err(); err != nil {
			closeClient(client)
			return nil, err
		}
		auths, err := authMethods(n)
		if err != nil {
			closeClient(client)
			return nil, err
		}
		var underlying net.Conn
		if i == 0 {
			underlying, err = dialer.DialContext(ctx, "tcp", n.hostport)
		} else {
			underlying, err = client.DialContext(ctx, "tcp", n.hostport) // 带 ctx(v0.48 签名)
		}
		if err != nil {
			closeClient(client)
			return nil, fmt.Errorf("hop %d dial %s: %w", i, n.hostport, err)
		}
		cc, chans, reqs, err := gossh.NewClientConn(underlying, n.hostport, &gossh.ClientConfig{
			User:            n.user,
			Auth:            auths,
			HostKeyCallback: d.hosts.CallbackFor(n.hostport),
			Timeout:         handshakeTimeout,
		})
		if err != nil {
			_ = underlying.Close()
			closeClient(client)
			return nil, fmt.Errorf("hop %d handshake %s: %w", i, n.hostport, err)
		}
		client = gossh.NewClient(cc, chans, reqs)
	}
	return client, nil
}

func closeClient(c *gossh.Client) {
	if c != nil {
		_ = c.Close()
	}
}
```

```go
package ssh

import (
	"context"
	"fmt"
	"sync"
	"time"

	gossh "golang.org/x/crypto/ssh"
)

// chain 是池里一条完整链(含跳板)。get 懒建;client 死亡由监控协程标记。
type chain struct {
	mu        sync.Mutex
	dialFn    func(ctx context.Context) (*gossh.Client, error)
	client    *gossh.Client
	clientErr error // 监控协程在 conn 死亡时写入
	lastFail  time.Time
	failCount int
	active    int // 当前经此链打开的 channel 数
}

func newChain(dialFn func(ctx context.Context) (*gossh.Client, error)) *chain {
	return &chain{dialFn: dialFn}
}

func chainBackoff(failCount int) time.Duration {
	d := time.Second << min(failCount, 5) // 1s,2s,…,30s 封顶
	if d > 30*time.Second {
		d = 30 * time.Second
	}
	return d
}

// get 返回存活 client;死了/没建则重建(受退避约束)。
func (c *chain) get(ctx context.Context) (*gossh.Client, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.client != nil && c.clientErr == nil {
		return c.client, nil
	}
	if c.client != nil {
		_ = c.client.Close()
		c.client = nil
	}
	if !c.lastFail.IsZero() {
		if wait := chainBackoff(c.failCount); time.Since(c.lastFail) < wait {
			return nil, fmt.Errorf("ssh chain rebuilding, backoff %s", wait)
		}
	}
	cl, err := c.dialFn(ctx)
	if err != nil {
		c.lastFail = time.Now()
		c.failCount++
		return nil, err
	}
	c.client = cl
	c.clientErr = nil
	c.failCount = 0
	// 监控:conn 一死立刻标记,下次 get 重建;不阻塞当前调用方。
	go func() {
		err := cl.Wait()
		c.mu.Lock()
		if c.client == cl { // 防止已被新 client 顶替时误标
			c.clientErr = err
		}
		c.mu.Unlock()
	}()
	return cl, nil
}

// drop 收走死链(仅当当前 client 已被标记死亡)。
func (c *chain) drop() {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.client != nil && c.clientErr != nil {
		_ = c.client.Close()
		c.client = nil
	}
}

func (c *chain) addActive() {
	c.mu.Lock()
	c.active++
	c.mu.Unlock()
}

func (c *chain) releaseActive() {
	c.mu.Lock()
	if c.active > 0 {
		c.active--
	}
	c.mu.Unlock()
}

// pool 是 N 条独立链的集合;每条新流量选活跃 channel 数最少的链。
type pool struct {
	mu     sync.Mutex
	chains []*chain
	next   int // 同分轮转
}

func newPool(size int, dialFn func(index int) func(ctx context.Context) (*gossh.Client, error)) *pool {
	if size < 1 {
		size = 1
	}
	p := &pool{}
	for i := 0; i < size; i++ {
		p.chains = append(p.chains, newChain(dialFn(i)))
	}
	return p
}

// acquire 选活跃数最少的链并取得 client;全部链都在退避窗口时返回错误。
// 已知取舍:选链在 p.mu 内、建链在 chain.mu 内,重建期间其余链不受锁串行
// (只有同一条链的重建互斥);「全部断时现建救急链」在语义上等价于「任一
// 链退避期满即重建」,池测试需钉住该行为。
func (p *pool) acquire(ctx context.Context) (*gossh.Client, *chain, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	var best *chain
	for i := 0; i < len(p.chains); i++ {
		idx := (p.next + i) % len(p.chains)
		c := p.chains[idx]
		if best == nil || c.active < best.active {
			best = c
		}
	}
	p.next = (p.next + 1) % len(p.chains)
	cl, err := best.get(ctx)
	if err != nil {
		return nil, nil, err
	}
	best.addActive()
	return cl, best, nil
}

func (p *pool) close() {
	p.mu.Lock()
	defer p.mu.Unlock()
	for _, c := range p.chains {
		c.mu.Lock()
		if c.client != nil {
			_ = c.client.Close()
			c.client = nil
		}
		c.mu.Unlock()
	}
}
```

```go
func New(final nodeConfig, poolSize int, jumps []nodeConfig, hostsDir string) (*SSH, error) {
	hosts := NewKnownHosts(filepath.Join(hostsDir, "known_hosts"))
	// fail-fast:解析一次终点与跳板的私钥

		hosts: hosts,
		final: final,
	}
	p.pool = newPool(poolSize, func(index int) func(ctx context.Context) (*gossh.Client, error) {
		return func(ctx context.Context) (*gossh.Client, error) {
			return d.dialChain(ctx)
		}
	})
	return p, nil
}

type directTCPIPReq struct {
	Addr       string
	Port       uint32
	OriginAddr string
	OriginPort uint32
}

func (s *SSH) DialContext(ctx context.Context, metadata *M.Metadata) (net.Conn, error) {
	req := directTCPIPReq{
		Addr:       metadata.HostName,
		Port:       uint32(metadata.DstPort),
		OriginAddr: "127.0.0.1",
		OriginPort: 0,
	}
	for attempt := 0; attempt < 2; attempt++ {
		cl, c, err := s.pool.acquire(ctx)
		if err != nil {
			return nil, fmt.Errorf("ssh acquire chain: %w", err)
		}
		ch, err := cl.OpenChannel("direct-tcpip", gossh.Marshal(req))
		if err == nil {
			return newTrackedChannel(ch, c), nil
		}
		// 换链重试仅限「client 已死」;MaxSessions 等业务错误透传(spec §4)
		c.drop()
		if attempt == 1 {
			return nil, fmt.Errorf("ssh open channel %s:%d: %w", metadata.HostName, metadata.DstPort, err)
		}
	}
	return nil, errors.New("ssh open channel: unreachable")
}

}

	err := t.Channel.Close()
	t.c.releaseActive()
	return err
}
```
